@RegisterTable("books", SCHEMA)
class Book extends Table {
  declare _id: string;

  @Field("string")
  @Relation({ to: () => Author })
  declare authorId: string;

  @Field("string")
  declare title: string;

  @Field("string")
  declare accountId: string;
}

  @Listable()
  @Access(AccessMode.ReadOnly)

  @Access(AccessMode.ReadWrite)
  declare authorId: string;

  @Listable()
  @Access(AccessMode.ReadWrite)
  @Sortable({ noIndex: true })
  declare title: string;

  @Listable()

  @Joined({ table: Author, localKey: "authorId", remoteField: "name" })
  declare authorName: string;

  @Listable()
  @Joined({ table: "accounts", localKey: "accountId", remoteField: "login", schema: AUTH_SCHEMA })
  declare accountLogin: string;
}

const lenientGuard = DefaultRoutes.WithGetGuard(function (this: DataControllerInstance) {
  return Promise.resolve(false) as unknown as Promise<void>;
});

const ownedGuard = DefaultRoutes.WithGetGuard(function (this: DataControllerInstance) {
  assert(this instanceof _ProbeAPI, 500, "guard without the controller");
});

const paramsGuard = DefaultRoutes.WithGetGuard(function (_ctx, _current, params) {
  assert(params.id === "p1", 500, "guard without the parsed parameters");
});

const alwaysThrowsGuard = DefaultRoutes.WithGetGuard(function () {
  assert(false, 500, "the guard ran");
});

const titleThisFilter: FilterFunction<Record<string, unknown>> = (context, proxy, _key, raw) =>
  proxy.eq(context.this instanceof _ProbeAPI ? raw : "__no_controller__");

const upperTagFilter: FilterFunction<Record<string, unknown>> = (_context, proxy, _key, raw) => proxy.eq(raw.toUpperCase());

@RegisterDataController()
class _ProbeAPI extends DataController(
  ProbeRow,
  {
    ...DefaultRoutes.All,
    lenient: lenientGuard,
    owned: ownedGuard,
    withParams: paramsGuard,
    alwaysThrows: alwaysThrowsGuard,
    plain: DefaultRoutes.WithOptions(DefaultRoutes.List, { pluckMode: "nothing", noPluck: true }),
    plainFalse: DefaultRoutes.WithOptions(DefaultRoutes.List, { pluckMode: "nothing", noPluck: false }),
    slugEdit: DefaultRoutes.WithOptions(DefaultRoutes.Edit, { index: "tag" }),
  },
  Controller("/probes"),
) {
  @ModelReference()
  @Model(ProbeModel)
  declare probeModel: ProbeModel;

  @Listable()
  @Access(AccessMode.ReadWrite)
  @Filter()
  declare kind: string;

  @Listable()
  @Access(AccessMode.ReadWrite)
  @Filter(titleThisFilter, false)
  declare title: string;

  @Listable()
  @Access(AccessMode.ReadWrite)
  @Filter(upperTagFilter)
  declare tag: string;

  @Listable()
  @Access(AccessMode.ReadWrite)
  @Filter()
  declare score: number;

  @Listable()
  @Access(AccessMode.ReadOnly)
  origin = "class-default";

  @Listable()
  @Access(AccessMode.ReadOnly)
  declare trace: string;

  @Access(AccessMode.WriteOnly)
  set mark(value: string | undefined) {
    const row: unknown = Reflect.get(this, "table");
    if (typeof row === "object" && row !== null) Reflect.set(row, "trace", value === undefined ? "setter-ran-for-absent" : `mark:${value}`);
  }
}

const overrideModes = {
  func: function (this: DataControllerInstance) {
    const meta = GetDataControllerMeta(this);
    const answer: Record<string, Record<string, number | undefined>> = {};
    for (const field of ["email", "name", "notes", "nickname"]) {
      answer[field] = {};
      for (const action of ["get", "list", "new", "edit"]) answer[field][action] = GetEffectiveMode(meta.fields[field], action);
    }
    return answer;
  },
  args: [Context()],
  method: "get",
};
