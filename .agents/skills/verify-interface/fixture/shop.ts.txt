const SCHEMA = "verify";
const AUTH_SCHEMA = "verify_auth";

const COMPARISONS = ["eq", "ne", "gt", "ge", "lt", "le"] as const;
type Comparison = (typeof COMPARISONS)[number];

function isComparison(mode: string): mode is Comparison {
  return COMPARISONS.some((comparison) => comparison === mode);
}

/** The numeric filter of docs/8.filters.md. */
const numberFilter: FilterFunction<Record<string, unknown>> = (_context, proxy, _key, raw, mode) => {
  const value = Number(raw);
  assert(raw.trim() !== "" && Number.isFinite(value), 400, "Invalid number");
  assert(isComparison(mode), 400, "Invalid comparison");
  return proxy[mode](value);
};

/** The email-domain filter of docs/8.filters.md: equality only, with the domain escaped. */
const emailDomainFilter: FilterFunction<Record<string, unknown>> = (_context, proxy, _key, raw, mode) => {
  assert(mode === "eq", 400, "Only equality is supported for email domains");
  const escaped = raw.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
  return proxy.match(`@${escaped}$`);
}

const EMAIL = /^[\w](\.?[\w-]+)*@(?:[a-zA-Z0-9]+(-[a-zA-Z0-9]+)*\.)+[a-zA-Z]{2,}$/;
const startsAtSchema = z.coerce.date();

const userDeclarationErrors = (): string[] => {
  const messages: string[] = [];
  try {
    DataController(class Bogus {}, {}, Controller("/bogus"));
    messages.push("declared");
  } catch (error) {
    messages.push(error instanceof Error ? error.message : "thrown");
  }
  return messages;
};

const UserExtras = {
  quickEdit: DefaultRoutes.WithOptions(DefaultRoutes.Edit, { noMandatory: "true" }),
  falseSkip: DefaultRoutes.WithOptions(DefaultRoutes.Edit, { noMandatory: "false" }),
  detailed: DefaultRoutes.WithOptions(DefaultRoutes.List, { pluckMode: "detailed" }),
  minimal: DefaultRoutes.WithOptions(DefaultRoutes.List, { pluckMode: "minimal" }),
  capped: DefaultRoutes.WithOptions(DefaultRoutes.List, { maxPage: 3 }),
  activeOnly: DefaultRoutes.WithOptions(DefaultRoutes.List, { filters: { status: ["active", "eq"] } }),
  unplucked: DefaultRoutes.WithOptions(DefaultRoutes.List, { noPluck: true }),
  missingMode: DefaultRoutes.WithOptions(DefaultRoutes.List, { pluckMode: "nothing" }),
  byEmail: DefaultRoutes.WithOptions(DefaultRoutes.Get, { index: "email" }),
  search: {
    func: async () => ({ results: [] }),
    args: [Context(), Parameters.List()],
    method: "get",
  },
  modes: {
    func: function (this: DataControllerInstance) {
      const meta = GetDataControllerMeta(this);
      const answer: Record<string, Record<string, number | undefined>> = {};
      for (const field of ["role", "age", "password", "_id"]) {
        answer[field] = {};
        for (const action of ["get", "list", "new", "edit"]) {
          answer[field][action] = GetEffectiveMode(meta.fields[field], action);
        }
      }
      return answer;
    },
    args: [Context()],
    method: "get",
  },
  declarations: {
    func: userDeclarationErrors,
    args: [],
    method: "get",
  },
};

@RegisterDataController()
class _UserAPI extends DataController(User, { ...DefaultRoutes.All, ...UserExtras }, Controller("/users")) {
  @ModelReference()
  @Model(UserModel)
  declare userModel: UserModel;

  @Listable()
  @Listable(true, "minimal")
  @Access(AccessMode.ReadOnly)
  @Sortable()
  declare _id: string;

  @Listable()
  @Access(AccessMode.ReadWrite)
  @Mandatory("new", "edit")
  @Validator((value) => typeof value === "string" && EMAIL.test(value))
  @Filter(emailDomainFilter, false)
  declare email: string;

  @Listable()
  @Listable(true, "detailed")
  @Listable(false, "minimal")
  @Access(AccessMode.ReadWrite)
  @Mandatory("new")
  @Sortable({ noIndex: true })
  @Filter()
  declare name: string;

  @Access(AccessMode.WriteOnly)
  @Mandatory("new")
  declare password: string;

  @Listable()
  @Access(AccessMode.ReadWrite, { edit: AccessMode.ReadOnly })
  @Filter()
  declare role: string;

  @Listable()
  @Access(AccessMode.ReadOnly, { new: AccessMode.ReadWrite })
  @Validator((value) => typeof value === "number" && value >= 0 && value <= 120)
  @Sortable({ noIndex: true })
  @Filter(numberFilter, false)
  declare age: number;

  @Listable()
  @Access(AccessMode.ReadWrite)
  @Mandatory("edit")
  @Filter()
  declare status: string;

  @Listable()
  @Access(AccessMode.ReadOnly)
  declare createdAt: string;

  @Access(AccessMode.ReadOnly)
  declare biography: string;

  @Listable(true, "detailed")
  @Access(AccessMode.ReadOnly)
  declare internalNotes: string;

  @Access(AccessMode.ReadWrite)
  @Validator(async (value) => typeof value === "string" && value !== "taken")
  declare nickname: string;

  @Access(AccessMode.ReadWrite)
  @Optional()
  declare notes: string;

  @Access(AccessMode.ReadWrite)
  @Validator((value) => startsAtSchema.safeParseAsync(value))
  declare startsAt: Date;

  @Listable()
  declare secretNote: string;
}

  @Listable()
  @Access(AccessMode.ReadWrite)
  @Foreign(User, "email")
  declare byEmail: User;

  @Listable()
  @Access(AccessMode.ReadWrite)
  @Foreign(OrderItem, undefined, true)
  declare itemsAll: OrderItem[];

  @Listable()

const invoiceGet = DefaultRoutes.WithGetGuard(async function (_ctx, current) {
  assert(current.documentType === "invoice", 404, "Not Found");
});

const stampedNew = {
  ...DefaultRoutes.New,
  func: function (this: DataControllerInstance, ctx: RequestContext, params: Parameters.NewParameters, body: Buffer) {
    const parsed = Validation.ParseBody(body);
    parsed.source = "import";
    return DefaultRoutes.New.func.call(this, ctx, params, parsed);
  },
};

const unstorableNew = {
  ...DefaultRoutes.New,
  func: function (this: DataControllerInstance, ctx: RequestContext, params: Parameters.NewParameters, body: Buffer) {
    const parsed = Validation.ParseBody(body);
    Reflect.set(parsed, "title", BigInt(1));
    return DefaultRoutes.New.func.call(this, ctx, params, parsed);
  },
};

@RegisterDataController()
class _DocumentAPI extends DataController(
  DocumentRow,
